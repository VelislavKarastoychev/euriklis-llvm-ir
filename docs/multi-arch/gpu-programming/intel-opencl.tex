\section{Vector Addition: Intel \acr{GPU} (\acr{OpenCL})}

\textbf{\acr{OpenCL} kernel:}
\begin{lstlisting}[style=ir]
__kernel void vectorAdd(__global float* a, __global float* b,
                        __global float* c, int n) {
  int idx = get_global_id(0);
  if (idx < n) {
    c[idx] = a[idx] + b[idx];
  }
}
\end{lstlisting}

\textbf{TypeScript with SPIRVCodegen:}
\begin{lstlisting}[style=ts]
import { SPIRVCodegen } from "@euriklis/llvm-ir";

const codegen = new SPIRVCodegen("vector_add");

// OpenCL provides global ID directly
const globalId = SPIRVCodegen.opencl.getGlobalIdX("gid");
// Note: Returns i64, may need ZExtInst for conversions

// Rest similar to CUDA/AMD versions
\end{lstlisting}

\textbf{Key \acr{OpenCL} intrinsics:}
\begin{itemize}[noitemsep]
  \item \texttt{SPIRVCodegen.opencl.getGlobalIdX/Y/Z}: Global work-item ID (i64)
  \item \texttt{SPIRVCodegen.opencl.getLocalIdX/Y/Z}: Local work-item ID
  \item \texttt{SPIRVCodegen.opencl.getGroupIdX/Y/Z}: Work-group ID
  \item \texttt{SPIRVCodegen.opencl.barrier()}: Work-group barrier
\end{itemize}
